(1) Create a Department Master Table 

Syntax:-

CREATE TABLE Dept
(
    dept_id   NUMBER(3) NOT NULL,
    dept_name VARCHAR2(50),
    Divn_id   NUMBER(3)
);


(2) Create Division Master Table in the following 

Syntax:-

CREATE TABLE Division_Master (
    Divn_id   NUMBER(3) PRIMARY KEY,
    Divn_name VARCHAR2(50),
    LOC       VARCHAR2(100)
);


(3) Create Employee information Table in the following 

Syntax:-

CREATE TABLE Employee_Info (
    Emp_id      NUMBER(10) PRIMARY KEY,
    Group_name  VARCHAR2(50),
    DOB         DATE,
    Mob_NO      VARCHAR2(15),
    email_id    VARCHAR2(100)
);

(4) Create Employ Details table in the following 

Syntax:-

CREATE TABLE Employ_Details (
    emp_id     NUMBER(5),
    divn_id    NUMBER(3),
    dept_id    NUMBER(3),
    Basic_Pay  NUMBER(10, 2)
);
